-- Description:	<Description,,>
-- =============================================
CREATE FUNCTION [reception].[GetList] 
(	
)
RETURNS TABLE 
AS
RETURN 
(
SELECT R.Id, R.Date, R.PurchaseId, R.Number, O.Date AS PurchaseDate, O.Number AS PurchaseNumber, ISNULL(R.DeliveryNumber, '') AS DeliveryNumber, R.DeliveryDate, O.SupplierId, S.CompanyName AS SupplierName, R.Status, 
                  O.BudgetArticle
FROM     reception.HEADERS AS R INNER JOIN
                  purchase.ORDERS AS O ON R.PurchaseId = O.Id INNER JOIN
                  dbo.SUPPLIERS AS S ON O.SupplierId = S.Id
)

CREATE PROCEDURE [dbo].[GetProductById]
    @ProductId uniqueidentifier
AS
BEGIN
    SET NOCOUNT ON;

    -- Première requête : Récupération des détails du produit
    SELECT
        p.Id,
        p.Designation,
        p.UnitMeasure,
        p.Quantity,
        p.TVA,
        p.DefaultWareHouseID,
        w.Name AS WareHouse
    FROM [INV].[dbo].[PRODUCTS] AS p
    full outer JOIN WareHouse AS w ON w.Id = p.DefaultWareHouseId
    WHERE p.Id = @ProductId;

    -- Deuxième requête : Vérification des réceptions existantes
    SELECT * 
    FROM reception.HEADERS AS rh
    WHERE EXISTS (
        SELECT 1 
        FROM reception.PRODUCTS AS r 
        WHERE r.ProductId = @ProductId
    ) 
    AND rh.Status = 1;
END;
GO

CREATE PROCEDURE [dbo].[GetReceptionDetails]
    @ReceptionId uniqueidentifier
AS
BEGIN

 -- Récupérer l'en-tête de la réception
    SELECT 
        h.Id,
        h.PurchaseId,
        h.Date,
        h.DeliveryNumber,
        h.DeliveryDate,
        h.Status
    FROM [INV].[reception].[HEADERS] h
    WHERE h.Id = @ReceptionId;

    -- Récupérer les produits de la réception
    SELECT 
        p.ReceptionId,
        p.ProductId,
        p.Quantity
    FROM [INV].[reception].[PRODUCTS] p
    WHERE p.ReceptionId = @ReceptionId;

   

    -- Récupérer la commande d'achat liée à la réception
    SELECT 
        o.Id,
        o.Number,
		s.CompanyName AS SupplierName,
        o.SupplierId,
        o.Date,
        o.BudgetArticle,
        o.BudgetType,
        o.ServiceType,
        o.TotalHT,
        o.TotalVA,
        o.TotalTC,
        o.CompletionDelay,
        o.VisaNumber,
        o.VisaDate,
        o.Status
    FROM [INV].[purchase].[ORDERS] o
	INNER JOIN dbo.SUPPLIERS AS s ON o.SupplierId = s.Id

    WHERE o.Id = (SELECT PurchaseId FROM [INV].[reception].[HEADERS] WHERE Id = @ReceptionId);
END;
GO

    JOIN [INV].[dbo].[PurchaseOrder] po ON po.IDSupplier = s.ID
    WHERE po.Number = @PurchaseOrderNumber;

    SELECT 
        od.Quantity,
        od.UnitPrice,
        od.TVA AS OrderDetailTVA,
        p.Designation,
        p.UnitMeasure,
        p.DefaultTVARate
    FROM [INV].[dbo].[OrderDetail] od
    JOIN [INV].[dbo].[Product] p ON od.IDProducts = p.ID
    JOIN [INV].[dbo].[PurchaseOrder] po ON od.IDPurchaseOrder = po.ID
    WHERE po.Number = @PurchaseOrderNumber;
END;

CREATE proc [reception].[CreateFromPurchase]
    @aPurchaseId uniqueIdentifier
as begin
    if not exists (select * from purchase.ORDERS where Id = @aPurchaseId)
        return 1001  -- not found

    declare @toDeliver int
    declare @receptionId uniqueidentifier
    select @toDeliver = sum(Quantity - Received) from purchase.PRODUCTS where PurchaseId = @aPurchaseId
    if @toDeliver = 0
        return 1002 -- all received
        if not exists(select * from purchase.PRODUCTS where PurchaseId=@aPurchaseId) 
        return 1003;

    set @receptionId  = newid()

    declare @purchaseNumber varchar(20),
    @supplierId uniqueidentifier,
    @supplierName varchar(200)

    select @purchaseNumber = Number, @supplierId = SupplierId from purchase.ORDERS where Id = @aPurchaseId

    select @supplierName = CompanyName from SUPPLIERS where Id = @supplierId

    select @receptionId as Id,null as Number, @aPurchaseId as PurchaseId, @purchaseNumber as purchaseNumber, 
        @supplierId as SupplierId, @supplierName as SupplierName, getdate() as Date, 
        null as DeliveryNumber, null as DeliveryDate, 0 as Status





                SELECT 
        @receptionId as ReceptionId,
        pp.ProductId,
        pp.Quantity,
        p.Designation,
        pp.Quantity-pp.Received as Received,
        p.DefaultWareHouseId as WareHouseId,
        pp.Quantity-pp.Received as Rest


    FROM [INV].purchase.[PRODUCTS] pp
    Inner Join [INV].[dbo].[PRODUCTS] as p on p.Id=pp.ProductId
         where pp.PurchaseId = @aPurchaseId 
                and pp.Quantity > pp.Received

end

CREATE PROCEDURE [reception].[GetById]
    @aReceptionId uniqueidentifier
AS
BEGIN
    declare 
        @purchaseId uniqueIdentifier,
        @purchaseNum varchar(50),
        @supplierId uniqueidentifier,
        @supplierName varchar(200)

    if not exists (select * from reception.HEADERS where Id = @aReceptionId)
        return 1001

    select @purchaseId = purchaseId from reception.HEADERS where Id = @aReceptionId

    select @purchaseNum = Number
    from purchase.ORDERS where Id = @purchaseId

    select @supplierId = SupplierId
    from purchase.ORDERS where Id = @purchaseId

    select @supplierName = CompanyName
    from SUPPLIERS where Id = @supplierId

    SELECT 
        h.Id,
        h.Number,
        h.PurchaseId,
        h.Date,
        @purchaseNum as PurchaseNumber,
        @supplierId as supplierId,
        @supplierName as supplierName,
        h.DeliveryNumber,
        h.DeliveryDate,
        h.Status
    FROM [INV].[reception].[HEADERS] h
    WHERE h.Id = @aReceptionId;

    -- Récupérer les produits de la réception
    SELECT 
        rp.ReceptionId,
        rp.ProductId,
        pp.Quantity,
        p.Designation,
        rp.Quantity as Received,
        p.DefaultWareHouseId as WareHouseId,
        pp.Quantity-pp.Received as Rest

    FROM [INV].[reception].[PRODUCTS] rp
    Inner Join [INV].[dbo].[PRODUCTS] as p on p.Id=rp.ProductId
    Inner Join [INV].purchase.[PRODUCTS] as pp on pp.ProductId=rp.ProductId

    WHERE rp.ReceptionId = @aReceptionId;




END;

CREATE PROCEDURE [reception].[Validate]
    @aReceiptId UNIQUEIDENTIFIER
AS
BEGIN
    SET NOCOUNT ON;

    if    (select Status from reception.HEADERS where Id = @aReceiptId) = 1
        return 2001

    if not exists (
        select *
        from reception.PRODUCTS as RP 
        inner join purchase.PRODUCTS as PP 
            ON RP.ProductId = PP.ProductId
        where RP.Quantity <= (PP.Quantity-PP.Received)
        )
        return 2002

    BEGIN TRAN;

        UPDATE reception.HEADERS
        SET Status = 1
        WHERE Id = @aReceiptId;

        DELETE FROM reception.PRODUCTS
        WHERE ReceptionId = @aReceiptId AND Quantity = 0;

        update purchase.PRODUCTS
            Set Received = Received + RP.Quantity
            from reception.PRODUCTS as RP 
            where RP.ReceptionId = @aReceiptId 
            and RP.ProductId = purchase.PRODUCTS.ProductId

    COMMIT TRAN;
END
